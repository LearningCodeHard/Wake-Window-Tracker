// Variables used by Scriptable.
// icon-color: green; icon-glyph: moon;
// Wake Window 2.1.1 — two-channel release support
const APP_VERSION = "2.1.1";
/* Wake Window V1 — pure scheduling, validation, and import functions. */
const WW = (() => {
  const defaults = {
    timezone: 'America/Indiana/Indianapolis', bedtime: '20:00',
    nap: {min: 60, ideal: 75, max: 90},
    first: {min: 60, ideal: 75, max: 90},
    second: {min: 120, ideal: 135, max: 150},
    later: {min: 120, ideal: 135, max: 150},
    final: {min: 120, ideal: 150, max: 180}
  };
  const copy = x => JSON.parse(JSON.stringify(x));
  const pad = n => String(n).padStart(2, '0');
  const clamp = (x, a, b) => Math.max(a, Math.min(b, x));
  function parts(value, tz) {
    const d = new Date(value);
    if (!Number.isFinite(+d)) throw Error('Invalid timestamp.');
    const p = {};
    new Intl.DateTimeFormat('en-US', {timeZone:tz, year:'numeric',month:'2-digit',day:'2-digit',hour:'2-digit',minute:'2-digit',hourCycle:'h23'}).formatToParts(d).forEach(x => p[x.type] = x.value);
    return {date:`${p.year}-${p.month}-${p.day}`, time:`${p.hour}:${p.minute}`};
  }
  function wall(date, time, tz) {
    if (!/^\d{4}-\d{2}-\d{2}$/.test(date) || !/^(?:[01]\d|2[0-3]):[0-5]\d$/.test(time)) throw Error('Use a valid date and 24-hour time.');
    const target = Date.parse(`${date}T${time}:00Z`);
    if (!Number.isFinite(target) || new Date(target).toISOString().slice(0,10) !== date) throw Error('Invalid calendar date.');
    let guess = target;
    for (let i=0; i<4; i++) {
      const p = parts(guess,tz);
      const represented = Date.parse(`${p.date}T${p.time}:00Z`);
      guess += target-represented;
    }
    const p = parts(guess,tz);
    if (p.date !== date || p.time !== time) throw Error('This local time does not exist because the clocks change.');
    for (const delta of [-120,-90,-60,-30,30,60,90,120]) {
      const other=parts(guess+delta*60000,tz);
      if (other.date===date && other.time===time) throw Error('This local time occurs twice because the clocks change. V1 needs an unambiguous daytime entry.');
    }
    return new Date(guess).toISOString();
  }
  function validateSettings(s) {
    if (!s || !/^(?:[01]\d|2[0-3]):[0-5]\d$/.test(s.bedtime)) throw Error('Enter a valid bedtime.');
    try { if(typeof s.timezone!=='string'||!s.timezone.trim())throw Error();parts(Date.now(),s.timezone); } catch { throw Error('Enter a valid IANA timezone, such as America/Indiana/Indianapolis.'); }
    for (const key of ['nap','first','second','later','final', ...['short','long'].filter(k=>s[k]!==undefined)]) {
      const r=s[key];
      if (!r || !['min','ideal','max'].every(k=>Number.isInteger(r[k]) && r[k]>=1 && r[k]<=600) || r.min>r.ideal || r.ideal>r.max) throw Error(`${key}: use whole minutes with minimum ≤ preferred ≤ maximum (1–600).`);
    }
    const {target,max}=napCounts(s);
    if(![target,max].every(n=>Number.isInteger(n)&&n>=0&&n<=6)||target>max)
      throw Error('Use a daily nap target and maximum between 0 and 6, with target at or below maximum.');
    return s;
  }
  const napCounts=s=>({target:s.napTarget===undefined?3:s.napTarget,max:s.napMax===undefined?6:s.napMax});
  const napProfiles=s=>({standard:s.nap,short:s.short||{min:20,ideal:35,max:45},long:s.long||{min:90,ideal:105,max:120}});
  function validateDay(day,s,now=Date.now()) {
    const naps=[...(day.naps||[])].sort((a,b)=>Date.parse(a.start)-Date.parse(b.start));
    let previous=day.morning ? Date.parse(day.morning) : -Infinity;
    if (day.morning && (!Number.isFinite(previous) || parts(previous,s.timezone).date!==day.date || previous>now)) throw Error('Morning wake must be on the selected day and not in the future.');
    let activeCount=0;
    for (const n of naps) {
      if(n.end===null){
        const a=Date.parse(n.start), b=Date.parse(n.expectedEnd);
        if(++activeCount>1)throw Error('More than one active nap was found. Cancel the extra active entry before planning.');
        if(!Number.isFinite(a)||a>now)throw Error('An active nap must have an actual start time, not a future start.');
        if(!Number.isFinite(b)||b<=a)throw Error('Expected wake must be after nap start.');
        if(parts(a,s.timezone).date!==day.date||parts(b,s.timezone).date!==day.date)throw Error('Keep the active nap and expected wake within the selected day.');
        if(a<previous)throw Error('Naps overlap, or a nap starts before morning wake. Correct the recorded times.');
        previous=Infinity;continue;
      }
      const a=Date.parse(n.start), b=Date.parse(n.end);
      if (!Number.isFinite(a) || !Number.isFinite(b) || b<=a) throw Error('Nap end must be after nap start.');
      if (parts(a,s.timezone).date!==day.date || parts(b,s.timezone).date!==day.date) throw Error('V1 records daytime naps within one calendar day.');
      if (a<previous) throw Error('Naps overlap, or a nap starts before morning wake. Correct the recorded times.');
      if (b>now) throw Error('Actual nap times cannot be in the future.');
      previous=b;
    }
    return naps;
  }
  function allocate(segments,total) {
    const lo=segments.reduce((a,s)=>a+s.min,0), hi=segments.reduce((a,s)=>a+s.max,0);
    if(total<lo || total>hi) return null;
    // Convex box-constrained least squares, then integer-minute apportionment.
    const scale=s=>Math.max(15,s.max-s.min);
    let left=-1e6,right=1e6;
    for(let n=0;n<70;n++) {
      const m=(left+right)/2;
      const sum=segments.reduce((a,s)=>a+clamp(s.ideal+m*scale(s)*scale(s),s.min,s.max),0);
      if(sum<total) left=m; else right=m;
    }
    const vals=segments.map(s=>Math.floor(clamp(s.ideal+((left+right)/2)*scale(s)*scale(s),s.min,s.max)+1e-7));
    let remain=total-vals.reduce((a,b)=>a+b,0);
    while(remain>0) {
      let best=-1,cost=Infinity;
      segments.forEach((s,i)=>{if(vals[i]<s.max){const c=(2*(vals[i]-s.ideal)+1)/(scale(s)*scale(s));if(c<cost){cost=c;best=i;}}});
      if(best<0) throw Error('Schedule allocation failed.');
      vals[best]++; remain--;
    }
    const score=segments.reduce((a,s,i)=>a+Math.pow((vals[i]-s.ideal)/scale(s),2),0)/segments.length;
    return {vals,score};
  }
  function planningAnchor(day,s,now=Date.now()) {
    validateSettings(s);
    const all=validateDay(day,s,now),naps=all.filter(n=>n.end!==null),activeNap=all.find(n=>n.end===null)||null;
    const activeInfo=activeNap?{activeNap,expectedWake:Date.parse(activeNap.expectedEnd),sleepElapsed:Math.max(0,Math.floor((now-Date.parse(activeNap.start))/60000))}:{};
    const anchor=activeNap?Date.parse(activeNap.expectedEnd):naps.length ? Date.parse(naps[naps.length-1].end) : (day.morning ? Date.parse(day.morning) : null);
    if(anchor===null) return {status:'empty',events:[],message:'Enter morning wake or import a completed nap to plan the day.'};
    const target=Date.parse(wall(day.date,day.bedtime||s.bedtime,s.timezone));
    if(activeNap&&now>=anchor)return {status:'active-overdue',events:[],anchor,target,elapsed:0,...activeInfo,message:'Still marked asleep. The expected wake time has passed. Update it or record the actual wake to refresh the remaining plan.'};
    const elapsed=Math.max(0,Math.ceil((now-anchor)/60000));
    return {anchor,target,elapsed,...activeInfo};
  }
  function plan(day,s,now=Date.now()) {
    const base=planningAnchor(day,s,now);
    if(base.status)return base;
    const {anchor,target,elapsed,activeNap,...activeDetails}=base;
    const naps=day.naps||[],activeInfo=activeNap?{activeNap,...activeDetails}:{};
    if(now>=target) return {status:'past',events:[],anchor,target,elapsed,...activeInfo,message:'The chosen bedtime has arrived. Actual records are kept; start a new day with a morning wake entry.'};
    const candidates=[];
    for(let count=0;count<=6;count++) {
      const seg=[];
      for(let i=0;i<count;i++) {
        const index=naps.length+i;
        const w=index===0?s.first:(index===1?s.second:s.later);
        seg.push({...w,kind:'awake'}, {...s.nap,kind:'nap'});
      }
      seg.push({...s.final,kind:'final'});
      const original={...seg[0]};
      const overdue=elapsed>seg[0].max;
      seg[0].min=Math.max(seg[0].min,elapsed);
      seg[0].max=Math.max(seg[0].max,elapsed);
      const low=seg.reduce((a,x)=>a+x.min,0), high=seg.reduce((a,x)=>a+x.max,0);
      const wanted=Math.round((target-anchor)/60000),total=clamp(wanted,low,high);
      const a=allocate(seg,total);
      let cursor=anchor;
      const events=seg.map((x,i)=>{const e={kind:x.kind,start:cursor,end:cursor+a.vals[i]*60000,minutes:a.vals[i]};cursor=e.end;return e;});
      candidates.push({count,events,bedtime:cursor,target,anchor,elapsed,overdue,original,...activeInfo,
        deviation:Math.round((cursor-target)/60000),score:a.score,
        slack:Math.min(total-low,high-total),earliest:anchor+low*60000,latest:anchor+high*60000});
    }
    candidates.sort((a,b)=>Math.abs(a.deviation)-Math.abs(b.deviation) || (Number(a.overdue)-Number(b.overdue)) || a.score-b.score || a.count-b.count);
    const result=candidates[0];
    result.status=result.deviation ? 'adjust' : (result.overdue?'overdue':'fit');
    result.message=result.deviation ? `No schedule reaches your chosen bedtime within the remaining duration limits. The nearest plan is ${Math.abs(result.deviation)} minutes ${result.deviation>0?'later':'earlier'}.` : (result.slack===0?'This plan fits exactly at a boundary of your limits. Any change will need a fresh calculation.':'This plan fits your bedtime and duration limits.');
    if(result.overdue) result.message+=' The current wake window has already exceeded its maximum; the plan assumes the next transition happens now.';
    if(activeNap)result.message+=' Remaining times assume the expected wake; the actual wake will recalculate them.';
    return result;
  }
  function parseOCR(text,now,tz,preferredMinutes=defaults.nap.ideal) {
    let t=String(text||'').replace(/[\u00a0\u202f]/g,' ').replace(/\r/g,'').trim();
    if(!t) return {error:'No text was received. Use a screenshot showing Start Time and End Time, or enter the nap manually.'};
    // Saved Huckleberry sleep details have no Start/End labels. Only accept
    // the bounded Details section, never clock values from the day behind it.
    const lines=t.split('\n').map(line=>line.trim()).filter(Boolean);
    const details=lines.map((line,i)=>/^details$/i.test(line)?i:-1).filter(i=>i>=0);
    if(details.length){
      if(details.length!==1)return {error:'More than one Details section was found. Open one completed nap and try again.'};
      const tail=lines.slice(details[0]+1),stop=tail.findIndex(line=>/^(?:notes|edit|done)$/i.test(line));
      const block=stop<0?tail:tail.slice(0,stop);
      const sleep=block[0]&&block[0].match(/^.+?\s+slept\s+for\s+((?:\d+\s*h(?:ours?|rs?)?\s*)?(?:\d+\s*m(?:in(?:ute)?s?)?)?)$/i);
      if(block.length!==3||!sleep||!sleep[1].trim())return {error:'Could not read a single completed sleep Details entry. Open its date and time range, or enter the nap manually.'};
      const dateText=block[1].match(/^(?:(?:Monday|Tuesday|Wednesday|Thursday|Friday|Saturday|Sunday),?\s+)?([A-Za-z]+)\s+(\d{1,2}),?\s+(\d{4})$/i);
      const months={january:1,jan:1,february:2,feb:2,march:3,mar:3,april:4,apr:4,may:5,june:6,jun:6,july:7,jul:7,august:8,aug:8,september:9,sep:9,sept:9,october:10,oct:10,november:11,nov:11,december:12,dec:12};
      if(!dateText||!months[dateText[1].toLowerCase()])return {error:'The sleep Details date was not readable. Use its English month/day/year date, or enter the nap manually.'};
      const date=`${dateText[3]}-${pad(months[dateText[1].toLowerCase()])}-${pad(Number(dateText[2]))}`;
      const range=block[2].match(/^(\d{1,2}:[0-5]\d\s*[ap]\.?\s*m\.?)\s*[-\u2013\u2014]\s*(\d{1,2}:[0-5]\d\s*[ap]\.?\s*m\.?)$/i);
      if(!range)return {error:'The sleep Details time range needs a readable start and end with AM/PM. Enter or confirm the times manually.'};
      // Reuse the existing timezone, future/overnight checks and duration review.
      t=`Start Time\n${date}, ${range[1]}\nEnd Time\n${date}, ${range[2]}\nDuration\n${sleep[1]}`;
    }
    const labels=[...t.matchAll(/\b(start\s*time|end\s*time)\b\s*:?/gi)];
    const startOnly=labels.length===1&&/^start/i.test(labels[0][1]);
    if(!startOnly&&(labels.length!==2 || !/^start/i.test(labels[0][1]) || !/^end/i.test(labels[1][1]))) return {error:'Could not identify a single labeled Start Time and optional End Time. Enter or confirm the times manually.'};
    function read(block) {
      const clock=block.match(/\b(\d{1,2}):([0-5]\d)(?::[0-5]\d)?\s*([ap]\.?\s*m\.?)?/i);
      if(!clock) throw Error('A labeled time was not readable.');
      let h=Number(clock[1]);const suffix=(clock[3]||'').replace(/[.\s]/g,'').toLowerCase();
      if(suffix){if(h<1||h>12)throw Error('Invalid 12-hour clock value.');h=h%12+(suffix==='pm'?12:0);}else if(h>23)throw Error('Invalid hour.');
      const before=block.slice(0,clock.index),dateMatch=before.match(/\b(\d{4}-\d{2}-\d{2})\b/);
      let date,assumed=false;
      if(/\byesterday\b/i.test(before)) {const noon=wall(parts(now,tz).date,'12:00',tz);date=parts(Date.parse(noon)-86400000,tz).date;}
      else if(/\btoday\b/i.test(before)) date=parts(now,tz).date;
      else if(dateMatch) date=dateMatch[1];
      else if(before.replace(/[\s,:]/g,'')==='') {date=parts(now,tz).date;assumed=true;}
      else throw Error('Unrecognized date text. V1 supports Today, Yesterday, or YYYY-MM-DD.');
      // Ambiguous 1–12 without AM/PM requires manual confirmation.
      if(!suffix && h>=1 && h<=12) assumed=true;
      return {iso:wall(date,`${pad(h)}:${clock[2]}`,tz),assumed};
    }
    try {
      const a=read(t.slice(labels[0].index+labels[0][0].length,startOnly?undefined:labels[1].index));
      if(startOnly){
        if(Date.parse(a.iso)>now)throw Error('The imported start time is in the future. Check the date and AM/PM.');
        const goal=t.match(/\b(?:goal(?:\s*time)?|target\s*wake(?:\s*time)?|expected\s*wake(?:\s*time)?)\s*:?\s*([^\n]*(?:\n[^\n]+)?)/i);
        const expectedEnd=goal?read(goal[1]).iso:new Date(Date.parse(a.iso)+preferredMinutes*60000).toISOString();
        if(Date.parse(expectedEnd)<=Date.parse(a.iso)||parts(expectedEnd,tz).date!==parts(a.iso,tz).date)throw Error('Check the expected wake time or enter it manually.');
        const running=/\b(?:running|in\s*progress|currently\s*sleeping|asleep)\b/i.test(t);
        const messages=[goal?'Expected wake extracted from the timer goal; please review.':'Expected wake uses your preferred nap duration; you can change it.'];
        if(!running)messages.push('No End Time was found. Confirm that this nap is still in progress.');
        if(a.assumed)messages.push('Check the date and AM/PM carefully.');
        return {kind:'active',start:a.iso,end:null,expectedEnd,estimateSource:goal?'goal':'preferred',date:parts(a.iso,tz).date,messages,needsActiveConfirmation:!running};
      }
      const b=read(t.slice(labels[1].index+labels[1][0].length));
      if(Date.parse(b.iso)<=Date.parse(a.iso)) throw Error('End time is not after start time. Check AM/PM and dates.');
      if(Date.parse(b.iso)>now) throw Error('The imported nap ends in the future. Check AM/PM and dates.');
      if(parts(a.iso,tz).date!==parts(b.iso,tz).date)throw Error('V1 imports daytime naps within one calendar day.');
      const messages=[];
      if(a.assumed||b.assumed) messages.push('A date or AM/PM marker was missing. Check both times carefully.');
      const duration=Math.round((Date.parse(b.iso)-Date.parse(a.iso))/60000);
      const dm=t.match(/\bduration\s*:?\s*(?:(\d+)\s*(?:h(?:ours?|rs?)?)\s*)?(?:(\d+)\s*(?:m(?:in(?:ute)?s?)?))\b/i);
      if(dm && (dm[1]||dm[2]) && Math.abs((Number(dm[1]||0)*60+Number(dm[2]||0))-duration)>2) messages.push('Displayed duration does not match the extracted start/end times.');
      return {start:a.iso,end:b.iso,date:parts(a.iso,tz).date,messages};
    } catch(e) {return {error:e.message};}
  }
  function napKey(start,end) {return `nap_${start.replace(/[^0-9]/g,'')}_${end===null?'active':end.replace(/[^0-9]/g,'')}`;}
  function reduce(records) {
    const groups={}, conflicts=[];
    for(const r of records) {if(r.schema!==1 || !r.entity || !r.id || !Number.isFinite(Date.parse(r.at))) throw Error('An unreadable event record was found. No records have been changed.');(groups[r.entity]||(groups[r.entity]=[])).push(r);}
    const latest={};
    for(const [entity,events] of Object.entries(groups)) {
      const byId=Object.fromEntries(events.map(e=>[e.id,e])),depths={};
      const depth=(e,seen=new Set())=>{if(depths[e.id]!==undefined)return depths[e.id];if(seen.has(e.id))throw Error('A cyclic revision history was found.');seen.add(e.id);return depths[e.id]=e.base?(byId[e.base]?depth(byId[e.base],seen)+1:1):0;};
      // A late duplicate original import must never replace a correction.
      events.sort((a,b)=>depth(a)-depth(b)||a.at.localeCompare(b.at)||a.id.localeCompare(b.id));
      latest[entity]=events[events.length-1];
      const winner=latest[entity];
      // Competing branches at the same revision depth need review.
      const siblings=events.filter(e=>depth(e)===depth(winner));
      if(siblings.length>1 && new Set(siblings.map(e=>JSON.stringify([e.deleted,e.value]))).size>1) conflicts.push(entity);
    }
    return {latest,conflicts};
  }
  return {defaults,copy,pad,parts,wall,validateSettings,validateDay,napCounts,napProfiles,planningAnchor,allocate,plan,parseOCR,napKey,reduce};
})();
if(typeof module!=='undefined' && module.exports) module.exports=WW;

/* Minute-resolution planner for wake windows, appointments and nap profiles. */
const WWPlanning = (() => {
  const W=typeof WW!=='undefined'?WW:require('./core.js');
  const minute=60000;
  const identity=n=>n.entity||n.start;
  const overlap=(a,b,c,d)=>a<d&&b>c;
  function validatePlanning(v,date,s){
    if(!v||!Array.isArray(v.awake)||!Array.isArray(v.pins)||v.awake.length>12||v.pins.length>6)throw Error('Planning data must contain up to 12 awake periods and 6 nap pins.');
    const ids=new Set(),ordinals=new Set();
    for(const x of [...v.awake,...v.pins]){if(!x||typeof x.id!=='string'||!x.id||x.id.length>100||ids.has(x.id))throw Error('Each planning entry needs a unique ID.');ids.add(x.id);}
    for(const a of v.awake){
      if(typeof a.label!=='string'||a.label.length>80)throw Error('Use an appointment label of at most 80 characters.');
      if(!Number.isFinite(Date.parse(a.start))||!Number.isFinite(Date.parse(a.end))||Date.parse(a.end)<=Date.parse(a.start))throw Error('Awake period end must be after start.');
      if(W.parts(a.start,s.timezone).date!==date||W.parts(a.end,s.timezone).date!==date)throw Error('Keep an awake period within its selected day.');
    }
    for(const p of v.pins){
      if(!Number.isInteger(p.ordinal)||p.ordinal<1||p.ordinal>50||ordinals.has(p.ordinal))throw Error('Only one pin is allowed per nap number.');ordinals.add(p.ordinal);
      if(!Array.isArray(p.anchors)||p.anchors.some(x=>typeof x!=='string')||p.anchors.length>=p.ordinal)throw Error('Nap pin history is invalid.');
      if(p.start!==null&&(!Number.isFinite(Date.parse(p.start))||W.parts(p.start,s.timezone).date!==date))throw Error('A pinned start must be on its selected day.');
      if(p.minutes!==null&&(!Number.isInteger(p.minutes)||p.minutes<1||p.minutes>600))throw Error('Pinned duration must be whole minutes between 1 and 600.');
      if(p.profile!==undefined&&p.profile!==null&&!Object.prototype.hasOwnProperty.call(W.napProfiles(s),p.profile))throw Error('Choose a valid nap profile.');
      if(p.start===null&&p.minutes===null&&!p.profile)throw Error('Pin a start, duration or profile, or release the pin.');
    }
    return v;
  }
  function alternatives(day,s,now=Date.now(),compare=false){
    const data=validatePlanning(day.planning||{awake:[],pins:[]},day.date,s);
    const base=W.planningAnchor(day,s,now),actual=W.validateDay(day,s,now),countActual=actual.length;
    const notes=[];
    for(const a of data.awake){
      if(actual.some(n=>n.end!==null&&overlap(Date.parse(n.start),Date.parse(n.end),Date.parse(a.start),Date.parse(a.end))))
        notes.push(`Recorded sleep overlaps ${a.label||'an awake period'}. Actual times are preserved.`);
    }
    const changedHistory=p=>p.anchors.some((id,i)=>!actual[i]||identity(actual[i])!==id);
    const pins=data.pins.filter(p=>p.ordinal>countActual||changedHistory(p));
    const blocked=(message,reason)=>[{...base,status:'blocked',events:[],message,reason,notes,pins}];
    if(day.planningConflict)return blocked('Both parents edited the day plan. Review the saved versions and choose the appointments and pins to use.','concurrent-plans');
    if(pins.some(changedHistory))return blocked('Earlier nap history changed. Release the affected pins and pin the new suggestions if needed.','history');
    if(pins.some(p=>p.start!==null&&Date.parse(p.start)<now))return blocked('A pinned nap start has passed. Log the nap, move its start, or release the pin.','expired-pin');
    if(base.activeNap&&data.awake.some(a=>overlap(Date.parse(base.activeNap.start),base.expectedWake,Date.parse(a.start),Date.parse(a.end)))){
      notes.push('The current nap or its expected wake overlaps an awake period.');
      if(base.status!=='active-overdue')return blocked('The current nap estimate overlaps a keep-awake period. Update expected wake or move the appointment.','active-overlap');
    }
    if(base.status)return [{...base,notes,pins}];
    const {anchor,target,elapsed}=base;
    const horizon=Math.floor((Date.parse(W.wall(day.date,'23:59',s.timezone))-anchor)/minute);
    const relative=t=>(Date.parse(t)-anchor)/minute;
    const periods=data.awake.filter(a=>Date.parse(a.end)>Math.max(now,anchor)).map(a=>({...a,start:relative(a.start),end:relative(a.end)}));
    const minimumBed=Math.ceil(Math.max(0,...periods.map(a=>a.end)));
    const wanted=Math.round((target-anchor)/minute);
    const {max,target:napTarget}=W.napCounts(s),limit=Math.max(0,max-countActual);
    const required=Math.max(0,...pins.map(p=>p.ordinal-countActual));
    if(required>limit)return blocked(`Nap ${countActual+required} is pinned but the daily maximum is ${max}. Release the pin or raise the maximum.`,'pin-count');
    const profiles=W.napProfiles(s);
    const late=day.morning&&W.parts(day.morning,s.timezone).time>='07:00';
    const wakeRange=index=>index===0?(late?s.second:s.first):index===1?s.second:s.later;
    const cost=(r,v)=>((v-r.ideal)/Math.max(15,r.max-r.min))**2;
    const event=(kind,start,end)=>({kind,start:anchor+start*minute,end:anchor+end*minute,minutes:end-start});
    const rank=(a,b)=>Math.abs(a.deviation)-Math.abs(b.deviation)||a.targetGap-b.targetGap||a.score-b.score||a.count-b.count;
    // For a given cursor and nap count, future constraints are independent of earlier durations.
    // Keep the cheapest path to each cursor, retaining links for the winning timeline.
    function solve(preferred){
      let states=new Map([[0,{score:0}]]),results=[];
      function keep(map,end,score,previous,kind,start,details){
        if(!map.has(end)||map.get(end).score>score)map.set(end,{score,previous,event:{...event(kind,start,end),...details}});
      }
      for(let count=0;count<=limit&&states.size;count++){
        let best;
        if(count>=required){
          for(const [cursor,state] of states){
            const low=Math.max(cursor+s.final.min,minimumBed,count===0?elapsed:0);
            const high=Math.min(cursor+s.final.max,horizon);
            if(low>high)continue;
            const end=Math.max(low,Math.min(high,wanted));
            const score=(state.score+cost(s.final,end-cursor))/(2*count+1);
            const deviation=end-wanted;
            if(!best||Math.abs(deviation)<Math.abs(best.deviation)||Math.abs(deviation)===Math.abs(best.deviation)&&score<best.score)
              best={cursor,state,end,score,deviation};
          }
        }
        if(best){
          const events=[event('final',best.cursor,best.end)];
          for(let node=best.state;node.event;node=node.previous)events.push(node.event);
          const totalNaps=countActual+count;
          results.push({...base,count,totalNaps,targetGap:Math.abs(totalNaps-napTarget),events:events.reverse(),
            bedtime:anchor+best.end*minute,deviation:best.deviation,score:best.score,notes,pins,
            status:best.deviation?'adjust':'fit'});
        }
        if(count===limit)break;
        const ordinal=countActual+count+1,pin=pins.find(p=>p.ordinal===ordinal);
        const wake=wakeRange(ordinal-1),awakeStates=new Map();
        for(const [cursor,state] of states){
          const fixed=pin&&pin.start!==null?relative(pin.start)-cursor:null;
          const lo=Math.max(wake.min,count===0?elapsed:0,fixed??0),hi=Math.min(wake.max,fixed??horizon);
          for(let m=lo;Number.isInteger(m)&&m<=hi&&cursor+m<=horizon;m++)
            keep(awakeStates,cursor+m,state.score+cost(wake,m),state,'awake',cursor);
        }
        const next=new Map();
        const choices=pin?.profile?[pin.profile]:pin&&pin.minutes!==null?['standard']:Object.keys(profiles);
        for(const [cursor,state] of awakeStates){
          const nextAppointment=periods.filter(a=>a.end>cursor).reduce((first,a)=>Math.min(first,a.start),horizon);
          for(const profile of choices){
            const range=profiles[profile],fixed=pin?.minutes??null;
            const lo=Math.max(range.min,fixed??0),hi=Math.min(range.max,fixed??horizon,nextAppointment-cursor,horizon-cursor);
            for(let m=lo;m<=hi;m++)keep(next,cursor+m,state.score+cost(range,m)+(profile===preferred?0:1),
              state,'nap',cursor,{ordinal,profile,pinned:!!pin});
          }
        }
        states=next;
      }
      return results.sort(rank);
    }
    const standard=solve('standard');
    if(!standard.length){
      if(elapsed>Math.max(s.final.max,wakeRange(countActual).max))
        return blocked('The current wake window already exceeds your configured maximum. Log any missed nap or review the applicable wake-window limits in Settings.','wake-limit');
      if(pins.length)return blocked('No schedule fits the saved nap pins the wake windows, nap profiles or appointments. Move or release a pin, or review its duration profile.','pins');
      if(periods.length)return blocked(`No schedule fits around ${periods.map(a=>a.label||'the keep-awake period').join(', ')} within your limits. Review the appointment times or nap and wake-window settings.`,'appointments');
      return blocked('No bedtime fits the remaining wake-window limits today. Review the recorded wake time, nap allowance or final wake-window settings.','day-end');
    }
    const best=standard[0],selected=[best];
    if(compare){
      const variants=standard.slice(1).concat(solve('long').slice(0,1),solve('short').slice(0,1));
      const diverse=(a,b)=>a.count!==b.count||Math.abs(a.bedtime-b.bedtime)>=15*minute||
        a.events.some((e,i)=>e.kind==='nap'&&Math.abs(e.minutes-b.events[i].minutes)>=15);
      // Include a different count when feasible, then the strongest distinct remaining option.
      const differentCount=standard.find(p=>p.count!==best.count);
      if(differentCount)selected.push(differentCount);
      for(const p of variants.sort(rank)){
        if(selected.length===3)break;
        if(selected.every(q=>diverse(p,q)))selected.push(p);
      }
      selected.sort((a,b)=>a===best?-1:b===best?1:rank(a,b));
    }
    return selected.map(p=>({...p,message:(p.deviation
      ?`The closest bedtime for this option is ${Math.abs(p.deviation)} minutes ${p.deviation>0?'later':'earlier'} than target.`
      :'This plan fits the bedtime target.')+(base.activeNap?' Remaining times use the expected wake.':'')}));
  }
  const plan=(day,s,now)=>alternatives(day,s,now)[0];
  W.validatePlanning=validatePlanning;W.napIdentity=identity;W.plan=plan;W.alternatives=alternatives;
  return W;
})();

const WWApproval = (() => {
  const W=typeof WW!=='undefined'?WW:require('./planning.js');
  const minute=60000;
  const byOrdinal=plan=>plan.events.filter(e=>e.kind==='nap');
  function constraints(day,s){
    return {
      morning:day.morning||null,bedtime:day.bedtime||s.bedtime,planning:day.planning,
      timezone:s.timezone,counts:W.napCounts(s),profiles:W.napProfiles(s),
      first:s.first,second:s.second,later:s.later,final:s.final
    };
  }
  function basis(S){
    const {day,settings}=S;
    return {
      constraints:constraints(day,settings),
      naps:day.naps.map(n=>({entity:W.napIdentity(n),revision:n.revision,start:n.start,end:n.end,expectedEnd:n.expectedEnd||null}))
    };
  }
  function approvalHeads(records){
    const superseded=new Set(records.flatMap(r=>[r.base,...(r.value?.resolved||[])]));
    return records.filter(r=>!superseded.has(r.id)).sort((a,b)=>b.at.localeCompare(a.at)||b.id.localeCompare(a.id));
  }
  const signature=S=>JSON.stringify([S.date,S.revisions,S.day.naps.map(n=>n.revision),S.approvalVersions?.map(v=>v.id)||[],S.conflicts]);
  function saveApproval(S,plan,generatedAt){
    const {anchor,target,bedtime,events,count,totalNaps,deviation}=plan;
    return {basis:basis(S),generatedAt,plan:{anchor,target,bedtime,events,count,totalNaps,deviation}};
  }
  function changes(S,approved){
    const current=basis(S),before=approved.basis,reasons=[];
    const keys={morning:'Morning wake changed.',bedtime:'The bedtime target changed.',planning:'Appointments or nap pins changed.',
      timezone:'The household timezone changed.',counts:'The daily nap target or maximum changed.',profiles:'Nap duration profiles changed.',
      first:'The first wake-window settings changed.',second:'The second wake-window settings changed.',
      later:'Later wake-window settings changed.',final:'The final wake-window settings changed.'};
    for(const [key,message] of Object.entries(keys)){
      if(JSON.stringify(before.constraints[key])!==JSON.stringify(current.constraints[key]))reasons.push(message);
    }
    const suggested=byOrdinal(approved.plan);
    const showDifference=(ordinal,label,value,planned)=>{
      if(!value||!planned)return;
      const delta=Date.parse(value)-new Date(planned).getTime();
      if(Math.abs(delta)>10*minute)reasons.push(`Nap ${ordinal} ${label} ${Math.round(Math.abs(delta)/minute)} minutes ${delta>0?'later':'earlier'} than approved.`);
    };
    for(let i=0;i<Math.max(before.naps.length,current.naps.length);i++){
      const baseline=before.naps[i],nap=current.naps[i],ordinal=i+1;
      if(baseline&&(!nap||baseline.entity!==nap.entity)){
        reasons.push(`The recorded nap history changed at nap ${ordinal}.`);
        continue;
      }
      const planned=suggested.find(e=>e.ordinal===ordinal);
      if(!baseline&&!planned){
        reasons.push(`Nap ${ordinal} was added beyond the approved schedule.`);
        continue;
      }
      showDifference(ordinal,'started',nap.start,baseline?.start||planned?.start);
      if(nap.end)showDifference(ordinal,'ended',nap.end,baseline?.end||baseline?.expectedEnd||planned?.end);
      else showDifference(ordinal,'is expected to end',nap.expectedEnd,baseline?.expectedEnd||planned?.end);
    }
    return reasons;
  }
  function overdue(S,approved,now){
    const next=byOrdinal(approved.plan).find(e=>e.ordinal>S.day.naps.length);
    if(next&&next.start<now)return `Nap ${next.ordinal} is overdue. Log it if it happened, or use Recalculate to review your options.`;
    const active=S.day.naps.find(n=>n.end===null);
    if(active&&Date.parse(active.expectedEnd)<=now)return 'Expected wake has passed. Update expected wake or record the actual wake.';
    return '';
  }
  function expired(plan,now){
    return plan.events.some(e=>e.kind==='nap'&&e.start<now||e.kind==='final'&&e.end<now);
  }
  W.approval={approvalHeads,signature,saveApproval,changes,overdue,expired};
  return W;
})();

      try{response={id:command.id,...await process(command),saveConfirmed:commandSaveConfirmed};}catch(e){response={id:command.id,error:e.message,lastSavedAt,saveConfirmed:commandSaveConfirmed};}
      if(closed)return;
      await web.evaluateJavaScript(`window.receive(${safeJSON(response)}); null;`,false);
    }
  }
  // Keep the script's main lifetime tied to presentation, even while a button
  // callback is pending. A pending callback must not prevent clean dismissal.
  const buttons=handleButtons().catch(async e=>{
    if(closed)return;
    bridgeError=e;
    try{await web.evaluateJavaScript(`connectionStopped(${safeJSON('Dashboard connection stopped. Close and reopen. '+e.message)}); null;`,false);}catch(_){}
  });
  try{await web.present(false);}finally{closed=true;}
  if(bridgeError)throw bridgeError;
  Script.complete();
}
